! =====================================================================

! LOAD CASE A: VERTICAL SKIN PULL

sensor_length = 1.00

t_skin   = 2.00

! Layer coordinates
y0 = 0
y1 = y0 + t_skin
y2 = y1 + t_buffer
y3 = y2 + t_pdms
y4 = y3 + t_sensor
y5 = y4 + t_encap

! ---------------------------------------------------------------------
! ELEMENT TYPE
! ---------------------------------------------------------------------

ET,1,PLANE183
KEYOPT,1,3,2          ! Plane strain
KEYOPT,1,6,1          ! Mixed u-P formulation

! ---------------------------------------------------------------------
! MATERIAL PROPERTIES
! ---------------------------------------------------------------------

! Material 1: Skin
E_skin  = 0.20
nu_skin = 0.49
MP,EX,1,E_skin
MP,PRXY,1,nu_skin

! Material 2: Buffer
E_buffer  = 0.10
nu_buffer = 0.49
MP,EX,2,E_buffer
MP,PRXY,2,nu_buffer

! Material 3: PDMS
E_pdms  = 1.80
nu_pdms = 0.49
MP,EX,3,E_pdms
MP,PRXY,3,nu_pdms

! Material 4: Gold sensor
E_sensor  = 110000
nu_sensor = 0.34
MP,EX,4,E_sensor
MP,PRXY,4,nu_sensor

! Material 5: Encapsulation
E_encap  = 1.80
nu_encap = 0.49
MP,EX,5,E_encap
MP,PRXY,5,nu_encap

! ---------------------------------------------------------------------
! GEOMETRY
! Split at X = sensor_length so the sensor occupies only the left region.
! ---------------------------------------------------------------------

! Skin
RECTNG,0,sensor_length,y0,y1
RECTNG,sensor_length,model_length,y0,y1

! Buffer
RECTNG,0,sensor_length,y1,y2
RECTNG,sensor_length,model_length,y1,y2

! PDMS
RECTNG,0,sensor_length,y2,y3
RECTNG,sensor_length,model_length,y2,y3

! Sensor-level region
! Left region becomes gold; right region becomes encapsulation.
RECTNG,0,sensor_length,y3,y4
RECTNG,sensor_length,model_length,y3,y4

! Top encapsulation
RECTNG,0,sensor_length,y4,y5
RECTNG,sensor_length,model_length,y4,y5

ALLSEL,ALL
AGLUE,ALL
NUMMRG,KP
NUMCMP,KP
NUMCMP,LINE
NUMCMP,AREA

SELTOL,tol

! ---------------------------------------------------------------------
! MATERIAL ASSIGNMENTS
! ---------------------------------------------------------------------

! Skin
ALLSEL,ALL
ASEL,S,LOC,Y,(y0+y1)/2
AATT,1,,1

! Buffer
ALLSEL,ALL
ASEL,S,LOC,Y,(y1+y2)/2
AATT,2,,1

! PDMS
ALLSEL,ALL
ASEL,S,LOC,Y,(y2+y3)/2
AATT,3,,1

! Gold sensor: left side only
ALLSEL,ALL
ASEL,S,LOC,Y,(y3+y4)/2
ASEL,R,LOC,X,0,sensor_length
AATT,4,,1

! Encapsulation beside the sensor: right side only
ALLSEL,ALL
ASEL,S,LOC,Y,(y3+y4)/2
ASEL,R,LOC,X,sensor_length,model_length
AATT,5,,1

! Top encapsulation
ALLSEL,ALL
ASEL,S,LOC,Y,(y4+y5)/2
AATT,5,,1

! ---------------------------------------------------------------------
! MESH
! ---------------------------------------------------------------------

ALLSEL,ALL
MSHAPE,0,2D

n_left   = 20

n_skin   = 20

ALLSEL,ALL
LSEL,S,LOC,Y,(y0+y1)/2
LESIZE,ALL,,,n_skin

ALLSEL,ALL
LSEL,S,LOC,Y,(y1+y2)/2
LESIZE,ALL,,,n_buffer

ALLSEL,ALL
LSEL,S,LOC,Y,(y2+y3)/2

ALLSEL,ALL
LSEL,S,LOC,Y,(y3+y4)/2

LSEL,S,LOC,Y,(y4+y5)/2
